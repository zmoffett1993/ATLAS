let skuData=[],locationRows=[],selectedAisle=8;

const normalize=v=>String(v??'').toUpperCase().replace(/\s+/g,'').replace(/[^A-Z0-9-]/g,'');
const esc=v=>String(v??'').replace(/[&<>"']/g,c=>({'&':'&amp;','<':'&lt;','>':'&gt;','"':'&quot;',"'":'&#39;'}[c]));
function strictMatch(sku,q){return normalize(sku).includes(normalize(q));}
async function fetchAll(table,select='*'){let all=[],offset=0;while(true){const r=await fetch(`${SUPABASE_URL}/rest/v1/${table}?select=${encodeURIComponent(select)}&order=created_at.asc&limit=1000&offset=${offset}`,{headers});if(!r.ok)throw new Error(`${table} ${r.status}`);const p=await r.json();all=all.concat(p);if(p.length<1000)return all;offset+=1000;}}
async function loadData(){try{const [skus,locs]=await Promise.all([fetchAll('skus','id,sku,description,active'),fetchAll('locations','id,sku_id,aisle,section,level_info,notes,verified')]);const byId=new Map(skus.filter(x=>x.active!==false).map(x=>[x.id,x]));locationRows=locs.filter(x=>byId.has(x.sku_id)).map(x=>({...x,sku:byId.get(x.sku_id).sku}));skuData=[...byId.values()].map(x=>({...x,locations:locationRows.filter(l=>l.sku_id===x.id).sort((a,b)=>Number(a.aisle)-Number(b.aisle)||String(a.section).localeCompare(String(b.section)))})).sort((a,b)=>a.sku.localeCompare(b.sku));$('status').textContent='LIVE';renderAisles();renderRecent();}catch(e){console.error(e);$('status').textContent='OFFLINE';$('status').classList.add('error');}}
function showView(name){Object.entries(views).forEach(([k,v])=>v.classList.toggle('hidden',k!==name));document.querySelectorAll('.nav-btn').forEach(b=>b.classList.toggle('active',b.dataset.view===name||(name==='map'&&b.dataset.view==='browse')||(name==='result'&&b.dataset.view==='home')));window.scrollTo({top:0,behavior:window.matchMedia('(prefers-reduced-motion: reduce)').matches?'auto':'smooth'});}
function recentItems(){try{return JSON.parse(localStorage.getItem('atlasRecent')||'[]')}catch{return[]}}
function saveRecent(sku){const now=Date.now();const next=[{sku,time:now},...recentItems().filter(x=>x.sku!==sku)].slice(0,6);localStorage.setItem('atlasRecent',JSON.stringify(next));renderRecent();}
function relativeTime(ts){const d=Date.now()-ts,m=Math.floor(d/60000),h=Math.floor(m/60),days=Math.floor(h/24);if(m<1)return 'Just now';if(m<60)return `${m} min ago`;if(h<24)return `${h} hr${h===1?'':'s'} ago`;if(days===1)return 'Yesterday';return `${days} days ago`;}
function renderRecent(){const list=recentItems();$('clearRecent').style.visibility=list.length?'visible':'hidden';$('recentList').innerHTML=list.length?list.map(x=>`<button class="recent-item" data-sku="${esc(x.sku)}"><svg viewBox="0 0 24 24" fill="none" stroke="currentColor" stroke-width="2"><circle cx="12" cy="12" r="8"/><path d="M12 7v5l3 2"/></svg><span><div class="recent-sku">${esc(x.sku)}</div><div class="recent-time">${esc(relativeTime(x.time))}</div></span><b>›</b></button>`).join(''):'<div class="empty-recent">Your recent SKU searches will appear here.</div>';document.querySelectorAll('.recent-item').forEach(b=>b.addEventListener('click',()=>selectSku(b.dataset.sku)));}
function showSuggestions(q){const box=$('suggestions');if(!normalize(q)){box.classList.add('hidden');return;}const matches=skuData.filter(x=>strictMatch(x.sku,q)).slice(0,10);box.innerHTML=matches.map(x=>`<div class="suggestion" data-sku="${esc(x.sku)}"><strong>${esc(x.sku)}</strong><span>${x.locations.length} location${x.locations.length===1?'':'s'}</span></div>`).join('');box.classList.toggle('hidden',!matches.length);box.querySelectorAll('.suggestion').forEach(s=>s.addEventListener('click',()=>selectSku(s.dataset.sku)));}
function mapMarkup(active=''){return `<div class="aisle-map"><div class="map-stack"><div class="map-section ${active==='A'?'active':''}">A</div><div class="driveway main">MAIN FORKLIFT DRIVEWAY</div><div class="map-section ${active==='B'?'active':''}">B</div><div class="driveway">FORKLIFT DRIVEWAY 2</div><div class="map-section ${active==='C'?'active':''}">C</div><div class="driveway">FORKLIFT DRIVEWAY 3</div><div class="map-section ${active==='D'?'active':''}">D</div></div><div class="direction"><span>Front<br>of Aisle</span><div class="direction-line"></div><span>Back<br>of Aisle</span></div></div>`;}
function selectSku(sku){const item=skuData.find(x=>x.sku===sku);if(!item)return;saveRecent(sku);$('searchInput').value=sku;$('suggestions').classList.add('hidden');$('clearSearch').classList.remove('hidden');renderResult(item);showView('result');}
function renderResult(item){const locs=item.locations||[];if(!locs.length){$('resultContent').innerHTML=`<div class="no-result"><strong>${esc(item.sku)}</strong><br>No warehouse location is assigned.</div>`;return;}const primary=locs[0],active=String(primary.section||'').toUpperCase();$('resultContent').innerHTML=`<div class="result-shell"><div class="result-header"><h1>${esc(item.sku)}</h1><span>${locs.length} LOCATION${locs.length===1?'':'S'} FOUND</span></div><div class="result-body"><div class="location-list">${locs.map(l=>`<div class="location-row"><div class="loc-badge">${esc(l.aisle)}${esc(l.section)}</div><div><h3>Aisle ${esc(l.aisle)} · Section ${esc(l.section)} · ${esc(l.level_info||'All Levels')}</h3><p>${locationDescription(l.section)}</p><strong>${l.verified?'Verified location':'Location pending verification'}</strong></div></div>`).join('')}</div><div class="mini-map">${mapMarkup(active)}</div></div></div>`;}
function locationDescription(section){return {A:'South of Main Forklift Driveway',B:'After Main Forklift Driveway',C:'After Forklift Driveway 2',D:'Back Section'}[String(section).toUpperCase()]||'Warehouse location';}
function renderAisles(){const loaded=[...new Set(locationRows.map(x=>Number(x.aisle)).filter(Number.isFinite))].sort((a,b)=>a-b);const aisles=loaded.length?loaded:Array.from({length:21},(_,i)=>i+1);$('aisleGrid').innerHTML=aisles.map(a=>`<button class="aisle-btn" data-aisle="${a}">${a}</button>`).join('');document.querySelectorAll('.aisle-btn').forEach(b=>b.addEventListener('click',()=>openAisle(Number(b.dataset.aisle))));}
function openAisle(a){selectedAisle=a;$('mapAisleTitle').textContent=`AISLE ${a}`;$('fullMap').innerHTML=mapMarkup('');showView('map');}
function openDrawer(){ $('drawer').classList.add('open');$('backdrop').classList.add('open');$('drawer').setAttribute('aria-hidden','false'); }
function closeDrawer(){ $('drawer').classList.remove('open');$('backdrop').classList.remove('open');$('drawer').setAttribute('aria-hidden','true'); }
